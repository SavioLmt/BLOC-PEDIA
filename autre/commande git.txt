######## 3. Commandes Git utiles à connaître ###########################
########################################################################
########################################################################

Commande			Fonction
git status			Voir les modifications
git log --oneline		Voir l'historique
git add .			Ajouter tous les fichiers modifiés
git commit -m "message"		Enregistrer les modifications
git push			Envoyer sur GitHub
git pull			Récupérer les modifications GitHub
git remote -v			Vérifier le dépôt distant
git branch			Voir les branches
git diff			Voir les modifications non indexées
